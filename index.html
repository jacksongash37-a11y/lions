function doPost(e) {
  try {
    var ss = SpreadsheetApp.openById("YOUR_SPREADSHEET_ID_HERE");
    var data = JSON.parse(e.postData.contents);
    var d = data.data;
    var sheetName = d.yob + " " + d.gender;
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) sheet = ss.insertSheet(sheetName);
    var headers = ["Name","YOB","Gender","Club","Jersey","AFL Position","Notes","Flag","Updated"];
    if (sheet.getLastRow() === 0) sheet.appendRow(headers);
    var rows = sheet.getLastRow() > 1 ?
      sheet.getRange(2,1,sheet.getLastRow()-1,3).getValues() : [];
    var rowIdx = -1;
    for (var i = 0; i < rows.length; i++) {
      if (rows[i][0]===d.name && String(rows[i][1])===String(d.yob) && rows[i][2]===d.gender) {
        rowIdx = i; break;
      }
    }
    var vals = [d.name,d.yob,d.gender,d.team,d.jersey,d.position,d.notes,d.flag,new Date()];
    if (rowIdx >= 0) sheet.getRange(rowIdx+2,1,1,9).setValues([vals]);
    else sheet.appendRow(vals);
    return ContentService.createTextOutput("ok");
  } catch(err) {
    return ContentService.createTextOutput("error: " + err.toString());
  }
}
